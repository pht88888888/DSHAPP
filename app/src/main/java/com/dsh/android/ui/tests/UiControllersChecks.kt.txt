private class Clock : UiTaskScheduler {
    var now = 100_000L
    private data class Task(val at: Long, val action: Runnable)
    private val tasks = mutableListOf<Task>()
    override fun postDelayed(task: Runnable, delayMs: Long) { tasks.add(Task(now + delayMs, task)) }
    override fun remove(task: Runnable) { tasks.removeAll { it.action === task } }
    fun advance(ms: Long) {
        val end = now + ms
        while (true) {
            val next = tasks.filter { it.at <= end }.minByOrNull { it.at } ?: break
            tasks.remove(next)
            now = next.at
            next.action.run()
        }
        now = end
    }
    fun snapshot() = tasks.map { it.action }
}

private var checks = 0
private suspend fun scenario(name: String, action: suspend () -> Unit) {
    action()
    checks++
    println("ok $checks - $name")
}

private class DownloadServer : AutoCloseable {
    private val pool = Executors.newCachedThreadPool { task -> Thread(task, "download-http-check").apply { isDaemon = true } }
    private val server = HttpServer.create(InetSocketAddress("127.0.0.1", 0), 0)
    private val started = listOf("a", "b", "headers").associateWith { CountDownLatch(1) }
    private val release = started.keys.associateWith { CountDownLatch(1) }
    val base: String get() = "http://127.0.0.1:${server.address.port}"
    init {
        server.executor = pool
        for (name in started.keys) server.createContext("/$name") { exchange ->
            try {
                val bytes = "$name-complete".repeat(400).toByteArray()
                if (name == "headers") {
                    started.getValue(name).countDown()
                    release.getValue(name).await(5, TimeUnit.SECONDS)
                    exchange.sendResponseHeaders(200, bytes.size.toLong())
                    exchange.responseBody.write(bytes)
                } else {
                    exchange.sendResponseHeaders(200, bytes.size.toLong())
                    exchange.responseBody.write(bytes, 0, 16); exchange.responseBody.flush()
                    started.getValue(name).countDown()
                    release.getValue(name).await(5, TimeUnit.SECONDS)
                    exchange.responseBody.write(bytes, 16, bytes.size - 16)
                }
            } catch (_: IOException) { // Cancellation may close the peer socket.
            } finally { exchange.close() }
        }
        server.start()
    }
    suspend fun awaitStart(name: String) = withContext(Dispatchers.IO) {
        check(started.getValue(name).await(3, TimeUnit.SECONDS)) { "HTTP request did not start: $name" }
    }
    fun finish(name: String) { release.getValue(name).countDown() }
    override fun close() { release.values.forEach { it.countDown() }; server.stop(0); pool.shutdownNow() }
}

private suspend fun awaitParts(directory: File, count: Int) = withTimeout(3_000) {
    while (directory.listFiles().orEmpty().count { it.name.endsWith(".part") } != count) delay(5)
}

                calls.add(Triple(name, method.name, values?.toList() ?: emptyList()))
                when (method.returnType) {
                    java.lang.Boolean.TYPE -> true
                    java.lang.Integer.TYPE -> 7
                    String::class.java -> "$name:${method.name}"
                    else -> null
                }
            })
        val activity = provider(ActivityPageBridgeActions::class.java, "activity")
        val bubble = provider(BubblePageBridgeActions::class.java, "bubble")
        val privilege = provider(PrivilegePageBridgeActions::class.java, "privilege")
        val dispatcher = NativePageBridgeDispatcher(activity, bubble, privilege)
        val bubbleNames = BubblePageBridgeActions::class.java.declaredMethods.map { it.name }.toSet()
        val privilegeNames = PrivilegePageBridgeActions::class.java.declaredMethods.map { it.name }.toSet()
        val defaults = setOf("bridgeCapabilities", "aiAnnotateRequest", "aiAnnotateAck")
        for (method in com.dsh.android.web.PageBridge::class.java.declaredMethods.filter { it.name !in defaults }) {
            val arguments = method.parameterTypes.map { if (it == java.lang.Boolean.TYPE) true else "payload" }.toTypedArray()
            val expected = when (method.name) { in bubbleNames -> "bubble"; in privilegeNames -> "privilege"; else -> "activity" }
            val result = method.invoke(dispatcher, *arguments)
            check(calls.last() == Triple(expected, method.name, arguments.toList()))
            if (method.returnType == String::class.java) check(result == "$expected:${method.name}")
            if (method.returnType == java.lang.Boolean.TYPE) check(result == true)
            if (method.returnType == java.lang.Integer.TYPE) check(result == 7)
        }
        check(calls.size == 44)
        check(dispatcher.peekAnnotation() == "activity:peekAnnotation")
        dispatcher.acknowledgeAnnotation("id")
        check(calls.last() == Triple("activity", "acknowledgeAnnotation", listOf("id")))
    }
    println("passed $checks production behavior checks")
